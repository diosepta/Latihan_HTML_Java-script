    <div class="calculator-container">
        <h1>Kalkulator Sederhana</h1>
        <p>Aplikasi kalkulator ini akan meminta Anda memasukkan angka dan operator menggunakan kotak prompt, lalu menampilkan hasilnya dalam kotak alert.</p>

        <button onclick="hitung()">Mulai Kalkulator</button>
    </div>

    <script>
        function hitung() {
            let angka1Input = prompt("Masukkan angka pertama:");
            // Jika pengguna membatalkan prompt, hentikan fungsi
            if (angka1Input === null) return;

            let operatorInput = prompt("Masukkan operator (+, -, *, /):");
            if (operatorInput === null) return;

            let angka2Input = prompt("Masukkan angka kedua:");
            if (angka2Input === null) return;

            // Konversi string ke angka
            let angka1 = parseFloat(angka1Input);
            let angka2 = parseFloat(angka2Input);

            // Validasi input
            if (isNaN(angka1) || isNaN(angka2)) {
                alert("Input tidak valid! Pastikan Anda memasukkan angka.");
                return;
            }

            let hasil;

            if (operatorInput === '+') {
                hasil = angka1 + angka2;
            } else if (operatorInput === '-') {
                hasil = angka1 - angka2;
            } else if (operatorInput === '*') {
                hasil = angka1 * angka2;
            } else if (operatorInput === '/') {
                if (angka2 === 0) {
                    alert("Error! Pembagian dengan nol tidak diizinkan.");
                    return;
                }
                hasil = angka1 / angka2;
            } else {
                alert("Error! Operator tidak valid. Gunakan +, -, *, atau /.");
                return;
            }

            alert("Hasilnya adalah: " + hasil);
        }
    </script>
